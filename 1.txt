🧩 STEP 1 — Flask Application
📄 app.py
from flask import Flask, render_template, request

app = Flask(__name__)

users = []

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/register', methods=['POST'])
def register():
    name = request.form['name']
    email = request.form['email']

    users.append([name, email])

    return "<h2>Registration Successful</h2><a href='/'>Go Back</a>"

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

const { Builder, By } = require('selenium-webdriver');

async function testLogin() {

    let driver = await new Builder().forBrowser('chrome').build();

    await driver.get("file:///C:/path-to-your/index.html");

    // Enter username
    await driver.findElement(By.id("user")).sendKeys("admin");

    // Enter password
    await driver.findElement(By.id("pass")).sendKeys("123");

    // Click login button
    await driver.findElement(By.tagName("button")).click();

    // Get result text
    let result = await driver.findElement(By.id("result")).getText();

    console.log("Result:", result);

    if(result === "Login Successful") {
        console.log("Test Passed");
    } else {
        console.log("Test Failed");
    }

    await driver.quit();
}

from flask import Flask, request, render_template

app = Flask(__name__)

users = []

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/register', methods=['POST'])
def register():
    name = request.form['name']
    email = request.form['email']

    if name == "" or email == "":

    users.append([name, email])

    return "<h3>Registration Successful</h3>"

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

🧩 STEP 4 — Kubernetes Deployment File
📄 deployment.yaml
apiVersion: apps/v1
kind: Deployment

metadata:
  name: auto-deployment

spec:
  replicas: 2

  selector:
    matchLabels:
      app: auto-app

  template:
    metadata:
      labels:
        app: auto-app

    spec:
      containers:
      - name: auto-container
        image: k8s-auto-app:v1
        ports:
        - containerPort: 5000

🔍 Important Points
replicas: 2 → 2 pods created 
Uses image v1 

🧩 STEP 5 — Service File
📄 service.yaml
apiVersion: v1
kind: Service

metadata:
  name: auto-service

spec:
  type: NodePort

  selector:
    app: auto-app

  ports:
  - port: 5000
    targetPort: 5000
    nodePort: 30008

🧩 STEP 6 — Apply Configuration
